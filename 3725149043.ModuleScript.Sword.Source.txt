			arg1:strike()
		end
	else
		if v_u4.midair then
			if tick() - arg1.lastLunge > 0.55 then
				arg1:spinAttack()
				return
			end
		elseif tick() - arg1.lastLunge >= 0.5 then
			arg1:swing()
		end
		return
	end
end
function v_u13.activated(arg1)
	-- upvalues: (copy) v_u4
	arg1.isHolding = true
	if arg1.tool.Sword == nil then
		return
	elseif v_u4.CanAct ~= false then
		arg1:fire()
	end
end
function v_u13.deactivated(arg1)
	arg1.isHolding = false
end
function v_u13.equipped(arg1, _)
	-- upvalues: (copy) v_u6, (copy) v_u2, (copy) v_u_sword_lunge_debounce, (copy) v_u7, (copy) v_u_run, (copy) v_u_contextaction, (copy) v_u8, (copy) v_u4, (ref) v_u10
	if arg1.player.Character == nil then
		return
	elseif arg1.tool.Sword ~= nil then
		if not arg1.Anims then
			arg1:setupAnims()
		end
		if not arg1.funcs then
			arg1.funcs = v_u6:getFunctions(arg1.tool.Name or arg1.data.baseTool)
		end
		if arg1.Anims.Idle then
			arg1.Anims.Idle:Play()
		end
		arg1.isEquipped = true
		local v_draw = arg1.tool.Handle.Draw
		local _ = arg1.player.Team.Name
		if v_draw:IsA("Attachment") then
			local v_getchildren = v_draw:GetChildren()
			v_draw = v_getchildren[math.random(1, #v_getchildren)]
		end
		v_u2:sendServer("PlaySoundObjectServer", v_draw, 30)
		if tick() - arg1.lastLunge < v_u_sword_lunge_debounce * arg1.reloadModifier then
			v_u7:set("Reload")
		else
			v_u7:set("Aim")
		end
		arg1.funcs.onEquip(arg1)
		if arg1.startC0 == nil and arg1.tool.Sword:FindFirstChild("Weld") then
			arg1.startC0 = arg1.tool.Sword.Weld.C0
		end
		if not arg1.holdCheck then
			arg1.holdCheck = v_u_run.Stepped:Connect(function()
				-- upvalues: (copy) arg1
				if tick() - arg1.lastSwing > 0.5 * arg1.swingSpeedModifier and arg1.isHolding then
					arg1:fire()
				end
			end)
		end
		v_u_contextaction:BindActionAtPriority("SwordSpecial", function(_, arg2, arg3)
			-- upvalues: (ref) v_u8, (ref) v_u4, (copy) arg1
			if v_u8.mouseLock and _G.Data.settings.altSpecial ~= 0 or arg3.UserInputType ~= Enum.UserInputType.MouseButton2 then
				if v_u4.CanAct ~= false then
					if arg2 == Enum.UserInputState.Begin then
						if arg1.tooldata.loopSpinAttack and v_u4.state == "Freefall" then
							arg1:midairLoopSpin()
							return
						end
						if arg1.tooldata.projectile then
							arg1:projectile()
							return
						end
						arg1:lunge()
					end
				end
			else
				return Enum.ContextActionResult.Pass
			end
		end, false, 1000, Enum.KeyCode.R, Enum.KeyCode.ButtonL2, Enum.UserInputType.MouseButton2)
		v_u2:send("ToolKey", "SpecialAttack", arg1.tooldata.specialAttack)
		local v1 = v_u8.inputType == "Touch" and arg1.player.PlayerScripts:FindFirstChild("TouchAction")
		if v1 then
			if v_u10 then
				v_u10:Disconnect()
			end
			v_u10 = v1.Event:Connect(function()
				-- upvalues: (ref) v_u4, (copy) arg1
				if v_u4.CanAct == false then
					return
				elseif arg1.tooldata.loopSpinAttack and v_u4.state == "Freefall" then
					arg1:midairLoopSpin()
					return
				elseif arg1.tooldata.projectile then
					arg1:projectile()
				else
					arg1:lunge()
				end
			end)
		end
	end
end
function v_u13.unequipped(arg1, _)
	-- upvalues: (ref) v_u10, (copy) v_u4, (copy) v_u7, (copy) v_u_contextaction, (copy) v_u2
	arg1.isHolding = false
	if arg1.holdCheck then
		arg1.holdCheck:Disconnect()
		arg1.holdCheck = nil
	end
	if arg1.Anims and arg1.Anims.Idle then
		arg1.Anims.Idle:Stop()
	end
	if v_u10 then
		v_u10:Disconnect()
	end
	if arg1.player.Character ~= nil then
		arg1.hasMesh = arg1.tool.Sword:FindFirstChild("Mesh")
		arg1.isEquipped = false
		for _, v1 in pairs(arg1.Anims) do
			v1:Stop()
		end
		v_u4.swordAction = nil
		arg1.funcs.onUnequip(arg1)
		v_u7:set()
		v_u_contextaction:UnbindAction("SwordSpecial")
		v_u2:send("ToolKey", "SpecialAttack", nil)
		v_u2:sendServer("DeactivateAllHitboxes")
	end
end

	-- upvalues: (copy) v_u13
